<script>
var currentStep=1;

function enhanceSelects(){
  var modal=document.getElementById('enrollModal');
  if(!modal||modal.dataset.csDone) return;
  modal.dataset.csDone='1';
  modal.querySelectorAll('select').forEach(function(sel){
    sel.style.display='none';
    var dd=document.createElement('div'); dd.className='cs-dd';
    var btn=document.createElement('button'); btn.type='button'; btn.className='cs-dd-btn';
    var panel=document.createElement('div'); panel.className='cs-dd-panel';
    var ddClose=document.createElement('button'); ddClose.type='button'; ddClose.className='cs-dd-x'; ddClose.setAttribute('aria-label','Close'); ddClose.innerHTML='&times;';
    ddClose.addEventListener('click',function(e){ e.stopPropagation(); dd.classList.remove('open'); });
    panel.appendChild(ddClose);
    function refreshBtn(){ var o=sel.options[sel.selectedIndex]; btn.textContent=o?o.textContent:''; btn.classList.toggle('placeholder',!sel.value); btn.classList.toggle('error',sel.classList.contains('error')); }
    function mkOpt(op){ var el=document.createElement('div'); el.className='cs-dd-opt'; el.textContent=op.textContent; el.dataset.val=op.value||op.textContent; el.addEventListener('click',function(e){ e.stopPropagation(); sel.value=op.value||op.textContent; sel.dispatchEvent(new Event('change',{bubbles:true})); dd.classList.remove('open'); panel.querySelectorAll('.cs-dd-opt').forEach(function(x){x.classList.remove('active');}); el.classList.add('active'); refreshBtn(); }); return el; }
    Array.from(sel.children).forEach(function(node){
      if(node.tagName==='OPTGROUP'){ var gl=document.createElement('div'); gl.className='cs-dd-optgroup'; gl.textContent=node.label; panel.appendChild(gl); Array.from(node.children).forEach(function(op){ panel.appendChild(mkOpt(op)); }); }
      else if(node.tagName==='OPTION'){ if(node.disabled&&node.hidden) return; panel.appendChild(mkOpt(node)); }
    });
    btn.addEventListener('click',function(e){ e.stopPropagation(); document.querySelectorAll('.cs-dd.open').forEach(function(x){ if(x!==dd) x.classList.remove('open'); }); dd.classList.toggle('open'); });
    sel.addEventListener('change',refreshBtn);
    dd.appendChild(btn); dd.appendChild(panel);
    sel.parentNode.insertBefore(dd,sel.nextSibling);
    refreshBtn();
  });
  document.addEventListener('click',function(e){ document.querySelectorAll('.cs-dd.open').forEach(function(x){ if(!x.contains(e.target)) x.classList.remove('open'); }); });
}
function enhanceDates(){
  var modal=document.getElementById('enrollModal');
  if(!modal||modal.dataset.dpDone) return;
  modal.dataset.dpDone='1';
  var MON=['January','February','March','April','May','June','July','August','September','October','November','December'];
  var DOW=['Su','Mo','Tu','We','Th','Fr','Sa'];
  modal.querySelectorAll('input[type="date"]').forEach(function(inp){
    inp.style.display='none';
    var dp=document.createElement('div'); dp.className='cs-dp';
    // typable text field (users can key the date in directly) + calendar icon
    var field=document.createElement('input'); field.type='text'; field.className='cs-dp-input';
    field.setAttribute('placeholder','DD/MM/YYYY'); field.setAttribute('inputmode','numeric'); field.setAttribute('autocomplete','off');
    var iconBtn=document.createElement('button'); iconBtn.type='button'; iconBtn.className='cs-dp-icon'; iconBtn.setAttribute('aria-label','Open calendar');
    iconBtn.innerHTML='<svg width="16" height="16" viewBox="0 0 24 24" fill="none" stroke-width="2" stroke-linecap="round" stroke-linejoin="round"><rect x="3" y="4" width="18" height="18" rx="3"/><line x1="16" y1="2" x2="16" y2="6"/><line x1="8" y1="2" x2="8" y2="6"/><line x1="3" y1="10" x2="21" y2="10"/></svg>';
    var cal=document.createElement('div'); cal.className='cs-dp-cal';
    var view=new Date(); view.setDate(1);
    var mode='days'; // 'days' | 'years'
    function parseAttr(a){ return a?new Date(a+'T00:00:00'):null; }
    var minD=parseAttr(inp.getAttribute('min')), maxD=parseAttr(inp.getAttribute('max'));
    function fmt(d){ return d.getFullYear()+'-'+String(d.getMonth()+1).padStart(2,'0')+'-'+String(d.getDate()).padStart(2,'0'); }
    function disp(d){ return String(d.getDate()).padStart(2,'0')+'/'+String(d.getMonth()+1).padStart(2,'0')+'/'+d.getFullYear(); }
    // Parse typed text: accepts DD/MM/YYYY, D-M-YYYY, D.M.YYYY, or YYYY-MM-DD
    function parseTyped(s){
      s=(s||'').trim(); if(!s) return null;
      var m=s.match(/^(\d{4})[-\/.](\d{1,2})[-\/.](\d{1,2})$/); // ISO YYYY-MM-DD
      var d,mo,y;
      if(m){ y=+m[1]; mo=+m[2]; d=+m[3]; }
      else { m=s.match(/^(\d{1,2})[-\/.](\d{1,2})[-\/.](\d{4})$/); if(!m) return null; d=+m[1]; mo=+m[2]; y=+m[3]; }
      if(mo<1||mo>12||d<1||d>31) return null;
      var dt=new Date(y,mo-1,d);
      if(dt.getFullYear()!==y||dt.getMonth()!==mo-1||dt.getDate()!==d) return null; // rejects e.g. 31/02
      return dt;
    }
    function refreshField(){
      minD=parseAttr(inp.getAttribute('min')); maxD=parseAttr(inp.getAttribute('max'));
      if(inp.value){ field.value=disp(new Date(inp.value+'T00:00:00')); }
      field.classList.toggle('error',inp.classList.contains('error'));
    }
    function commitTyped(){
      var dt=parseTyped(field.value);
      if(!field.value.trim()){ inp.value=''; inp.dispatchEvent(new Event('change',{bubbles:true})); return; }
      if(dt && (!minD||dt>=minD) && (!maxD||dt<=maxD)){
        inp.value=fmt(dt); view=new Date(dt); view.setDate(1);
        field.classList.remove('error'); inp.classList.remove('error');
        field.value=disp(dt);
        if(typeof clearError==='function') clearError(inp);
      } else {
        // decide a helpful, specific message
        var msg='Please enter a valid date (DD/MM/YYYY)';
        if(dt && maxD && dt>maxD) msg='Date of birth cannot be in the future';
        else if(dt && minD && dt<minD) msg='Please enter a more recent date';
        field.classList.add('error');
        if(typeof setError==='function') setError(inp,msg);
      }
      inp.dispatchEvent(new Event('change',{bubbles:true}));
    }
    function renderDays(){
      var y=view.getFullYear(), m=view.getMonth();
      var first=new Date(y,m,1).getDay(), dim=new Date(y,m+1,0).getDate();
      var sel=inp.value?new Date(inp.value+'T00:00:00'):null;
      // disable prev arrow if the whole previous month is before min; next arrow if whole next month is after max
      var prevLast=new Date(y,m,0);       // last day of previous month
      var nextFirst=new Date(y,m+1,1);    // first day of next month
      var prevDis=minD&&prevLast<minD, nextDis=maxD&&nextFirst>maxD;
      var h='<button type="button" class="cs-dp-x" data-dpclose="1" aria-label="Close">&times;</button><div class="cs-dp-head"><button type="button" class="cs-dp-nav'+(prevDis?' disabled':'')+'" data-d="-1"'+(prevDis?' disabled':'')+'>&#8249;</button><span class="cs-dp-title" data-yearview="1">'+MON[m]+' '+y+'</span><button type="button" class="cs-dp-nav'+(nextDis?' disabled':'')+'" data-d="1"'+(nextDis?' disabled':'')+'>&#8250;</button></div><div class="cs-dp-grid">';
      DOW.forEach(function(d){ h+='<div class="cs-dp-dow">'+d+'</div>'; });
      for(var i=0;i<first;i++) h+='<div class="cs-dp-day empty"></div>';
      for(var day=1;day<=dim;day++){
        var cur=new Date(y,m,day); var dis=(minD&&cur<minD)||(maxD&&cur>maxD);
        var isSel=sel&&sel.getFullYear()===y&&sel.getMonth()===m&&sel.getDate()===day;
        h+='<div class="cs-dp-day'+(dis?' disabled':'')+(isSel?' sel':'')+'" data-day="'+day+'">'+day+'</div>';
      }
      h+='</div><div class="cs-dp-foot"><button type="button" data-act="clear">Clear</button><button type="button" data-act="today">Today</button></div>';
      cal.innerHTML=h;
    }
    function renderYears(){
      var y=view.getFullYear();
      var minY=minD?minD.getFullYear():1950;
      var maxY=maxD?maxD.getFullYear():(new Date().getFullYear());
      var selY=inp.value?new Date(inp.value+'T00:00:00').getFullYear():null;
      var h='<button type="button" class="cs-dp-x" data-dpclose="1" aria-label="Close">&times;</button><div class="cs-dp-head"><span class="cs-dp-title" data-dayview="1">'+MON[view.getMonth()]+' '+y+'</span></div><div class="cs-dp-years">';
      // newest year first so recent birth years are at the top
      for(var yr=maxY; yr>=minY; yr--){
        h+='<div class="cs-dp-yr'+(yr===selY?' sel':'')+'" data-yr="'+yr+'">'+yr+'</div>';
      }
      h+='</div>';
      cal.innerHTML=h;
      // scroll selected year into view
      var s=cal.querySelector('.cs-dp-yr.sel'); if(s) s.scrollIntoView({block:'center'});
    }
    function render(){ minD=parseAttr(inp.getAttribute('min')); maxD=parseAttr(inp.getAttribute('max')); if(mode==='years') renderYears(); else renderDays(); }
    cal.addEventListener('click',function(e){
      e.stopPropagation();
      var closeBtn=e.target.closest('[data-dpclose]'); if(closeBtn){ dp.classList.remove('open'); return; }
      var titleY=e.target.closest('[data-yearview]'); if(titleY){ mode='years'; render(); return; }
      var titleD=e.target.closest('[data-dayview]'); if(titleD){ mode='days'; render(); return; }
      var yr=e.target.closest('.cs-dp-yr'); if(yr&&yr.dataset.yr){ view.setFullYear(parseInt(yr.dataset.yr,10)); mode='days'; render(); return; }
      var nav=e.target.closest('.cs-dp-nav'); if(nav){ if(nav.classList.contains('disabled')) return; view.setMonth(view.getMonth()+parseInt(nav.dataset.d,10)); render(); return; }
      var act=e.target.closest('[data-act]'); if(act){ if(act.dataset.act==='clear'){ inp.value=''; field.value=''; field.classList.remove('error'); } else { var td=new Date(); if((!minD||td>=minD)&&(!maxD||td<=maxD)){ inp.value=fmt(td); view=new Date(td); view.setDate(1);} } inp.dispatchEvent(new Event('change',{bubbles:true})); refreshField(); render(); if(act.dataset.act!=='clear') dp.classList.remove('open'); return; }
      var day=e.target.closest('.cs-dp-day'); if(day&&day.dataset.day){ inp.value=fmt(new Date(view.getFullYear(),view.getMonth(),parseInt(day.dataset.day,10))); inp.dispatchEvent(new Event('change',{bubbles:true})); refreshField(); render(); dp.classList.remove('open'); }
    });
    function openCal(){ document.querySelectorAll('.cs-dp.open').forEach(function(x){ if(x!==dp) x.classList.remove('open'); }); if(inp.value){view=new Date(inp.value+'T00:00:00');view.setDate(1);} mode='days'; render(); dp.classList.add('open'); }
    iconBtn.addEventListener('click',function(e){ e.stopPropagation(); if(dp.classList.contains('open')) dp.classList.remove('open'); else openCal(); });
    field.addEventListener('focus',function(){ openCal(); });
    field.addEventListener('click',function(e){ e.stopPropagation(); });
    // auto-insert the "/" separators as the user types (DD/MM/YYYY)
    field.addEventListener('input',function(e){
      // leave ISO-style typing (with a dash) alone; only auto-format digit entry
      if(field.value.indexOf('-')!==-1) return;
      var atEnd=(field.selectionStart===field.value.length);
      var digits=field.value.replace(/\D/g,'').slice(0,8);
      var out=digits;
      if(digits.length>4){ out=digits.slice(0,2)+'/'+digits.slice(2,4)+'/'+digits.slice(4); }
      else if(digits.length>2){ out=digits.slice(0,2)+'/'+digits.slice(2); }
      field.value=out;
      if(atEnd){ field.setSelectionRange(out.length,out.length); }
    });
    field.addEventListener('keydown',function(e){ if(e.key==='Enter'){ e.preventDefault(); commitTyped(); dp.classList.remove('open'); } });
    field.addEventListener('blur',function(e){ if(cal.contains(e.relatedTarget)) return; commitTyped(); });
    inp.addEventListener('change',refreshField);
    dp.appendChild(field); dp.appendChild(iconBtn); dp.appendChild(cal);
    inp.parentNode.insertBefore(dp,inp.nextSibling);
    render(); refreshField();
  });
  document.addEventListener('click',function(e){ document.querySelectorAll('.cs-dp.open').forEach(function(x){ if(!x.contains(e.target)) x.classList.remove('open'); }); });
}
function enrollFromMenu(){
  /* hard-close the mobile menu directly by ID (robust) */
  ['mobileMenu','hamburger','mobileBackdrop'].forEach(function(id){ var el=document.getElementById(id); if(el) el.classList.remove('open'); });
  openEnrollModal(); /* this re-applies the scroll lock and shows the form */
}
function resetEnrollForm(){
  /* Hide success screen, show footer & dots */
  document.getElementById('successScreen').classList.remove('show');
  document.getElementById('modalFooter').style.display='';
  document.getElementById('stepDots').style.display='';

  /* Clear all inputs, selects, textareas */
  document.querySelectorAll('#enrollModal input:not([type="checkbox"]):not([type="radio"])').forEach(function(el){ el.value=''; el.classList.remove('error'); });
  document.querySelectorAll('#enrollModal select').forEach(function(el){ el.selectedIndex=0; el.classList.remove('error'); el.dispatchEvent(new Event('change',{bubbles:true})); });
  document.querySelectorAll('#enrollModal textarea').forEach(function(el){ el.value=''; el.classList.remove('error'); });
  document.querySelectorAll('#enrollModal input[type="checkbox"]').forEach(function(el){ el.checked=false; });
  var pErrR=document.getElementById('programsErr'); if(pErrR) pErrR.style.display='none';
  document.querySelectorAll('#enrollModal .radio-item').forEach(function(el){ el.classList.remove('selected'); });
  document.querySelectorAll('#enrollModal .check-item').forEach(function(el){ el.classList.remove('checked'); });
  document.getElementById('consentItem').style.borderColor='';
  /* consent is pre-agreed by default */
  var cons=document.getElementById('consent'); if(cons){ cons.checked=true; var ci=document.getElementById('consentItem'); if(ci) ci.classList.add('checked'); }
  /* release the submit lock + re-enable the button for a fresh submission */
  window._nestSubmitting=false;
  var _sb=document.getElementById('btnSubmit'); if(_sb){ _sb.disabled=false; _sb.style.opacity=''; _sb.style.pointerEvents=''; }

  /* Reset custom (styled) dropdown buttons back to their placeholder */
  document.querySelectorAll('#enrollModal .cs-dd').forEach(function(dd){
    var sel=dd.previousElementSibling; // the hidden native <select>
    var btn=dd.querySelector('.cs-dd-btn');
    if(sel&&btn){ var o=sel.options[sel.selectedIndex]; btn.textContent=o?o.textContent:''; btn.classList.toggle('placeholder',!sel.value); btn.classList.remove('error'); }
    dd.querySelectorAll('.cs-dd-opt').forEach(function(x){x.classList.remove('active');});
    dd.classList.remove('open');
  });
  /* Reset custom date picker visible field */
  document.querySelectorAll('#enrollModal .cs-dp .cs-dp-input').forEach(function(f){ f.value=''; f.classList.remove('error'); });

  /* Go back to step 1 */
  goToStep(1);
}
function openEnrollModal(){
  document.getElementById('enrollModal').classList.add('open');
  document.body.style.overflow='hidden';document.documentElement.style.overflow='hidden';document.documentElement.classList.add('nest-modal-open');
  enhanceSelects();
  var today=new Date().toISOString().split('T')[0];
  document.getElementById('prefDate').min=today;
  document.getElementById('childDob').max=today;
  enhanceDates();
  resetEnrollForm();
}
function closeEnrollModal(){
  if(window._nestResetTimer){ clearTimeout(window._nestResetTimer); window._nestResetTimer=null; }
  document.getElementById('enrollModal').classList.remove('open');
  document.body.style.overflow='';document.documentElement.style.overflow='';document.documentElement.classList.remove('nest-modal-open');
  resetEnrollForm();
}

document.addEventListener('DOMContentLoaded',function(){
  /* background click does NOT close the form — only the X button, so users do not lose entered data */

  /* ── Alpha fields: only letters/spaces/hyphens, max enforced by HTML ── */
  ['childFirst','childLast','parentName'].forEach(function(id){
    var el=document.getElementById(id);
    if(!el) return;
    el.addEventListener('input',function(){
      /* strip anything that is not a letter (Latin or Urdu) space hyphen apostrophe */
      this.value=this.value.replace(/[^a-zA-Z\u0600-\u06FF\s\-']/g,'');
      clearError(this);
    });
    el.addEventListener('blur',function(){ liveAlpha(this); });
  });

  /* ── Phone field: digits / dash / plus only, max 15 enforced by HTML ── */
  var phoneEl=document.getElementById('parentPhone');
  if(phoneEl){
    phoneEl.setAttribute('inputmode','tel');
    phoneEl.addEventListener('input',function(){
      this.value=this.value.replace(/[^0-9\-+\s]/g,'');
      clearError(this);
    });
    phoneEl.addEventListener('blur',function(){ livePhone(this); });
  }

  /* ── Date fields ── */
  ['childDob','prefDate'].forEach(function(id){
    var el=document.getElementById(id);
    if(el) el.addEventListener('change',function(){ if(this.value) clearError(this); });
  });

  /* ── Select fields ── */
  ['childGender','childDiag','parentRel','parentArea','prefTime'].forEach(function(id){
    var el=document.getElementById(id);
    if(el) el.addEventListener('change',function(){ if(this.value) clearError(this); });
  });

  /* ── Email ── */
  var emailEl=document.getElementById('parentEmail');
  if(emailEl){
    emailEl.addEventListener('input',function(){
      if(this.value.length===0){ clearError(this); return; }
      if(!/^[^@\s]+@[^@\s]+\.[^@\s]{2,}$/.test(this.value)) setError(this);
      else clearError(this);
    });
  }

  /* ── Program checkboxes ── */
  document.querySelectorAll('#programsGroup input').forEach(function(cb){
    cb.addEventListener('change',function(){
      if(document.querySelectorAll('#programsGroup input:checked').length>0)
        var pErr2=document.getElementById('programsErr'); if(pErr2) pErr2.style.display='none';
    });
  });

  /* ── Consent ── */
  var con=document.getElementById('consent');
  if(con) con.addEventListener('change',function(){
    if(this.checked) document.getElementById('consentItem').style.borderColor='';
  /* consent is pre-agreed by default */
  var cons=document.getElementById('consent'); if(cons){ cons.checked=true; var ci=document.getElementById('consentItem'); if(ci) ci.classList.add('checked'); }
  /* release the submit lock + re-enable the button for a fresh submission */
  window._nestSubmitting=false;
  var _sb=document.getElementById('btnSubmit'); if(_sb){ _sb.disabled=false; _sb.style.opacity=''; _sb.style.pointerEvents=''; }
  });
});

/* ── helpers ── */
function errMsgEl(el){ var w=el.parentNode; return w?w.querySelector('.err-msg'):null; }
function setError(el,msg){
  el.classList.add('error');
  var m=errMsgEl(el);
  if(m){ if(!m.dataset.default) m.dataset.default=m.textContent; if(msg) m.textContent=msg; }
}
function clearError(el){
  el.classList.remove('error');
  var m=errMsgEl(el);
  if(m && m.dataset.default) m.textContent=m.dataset.default;
}

function liveAlpha(el){
  var v=el.value.trim();
  if(!v || v.length<2){ setError(el); return false; }
  if(!/^[a-zA-Z\u0600-\u06FF\s\-']+$/.test(v)){ setError(el); return false; }
  clearError(el); return true;
}
function livePhone(el){
  var v=el.value.trim();
  /* Pakistan numbers: 10-13 digits (with optional +92 prefix) */
  if(!v || v.length<10){ setError(el); return false; }
  if(!/^(\+92|0)[0-9\-\s]{9,13}$/.test(v)){ setError(el); return false; }
  clearError(el); return true;
}

/* ── step navigation ── */
function goToStep(n){
  document.querySelectorAll('.step-panel').forEach(function(p){p.classList.remove('active')});
  var panel=document.getElementById('step-'+n);
  if(panel) panel.classList.add('active');
  for(var i=1;i<=4;i++){
    var d=document.getElementById('dot-'+i);
    if(!d) continue;
    d.classList.remove('active','done');
    if(i===n) d.classList.add('active');
    else if(i<n) d.classList.add('done');
  }
  document.getElementById('btnBack').style.visibility=n>1?'visible':'hidden';
  document.getElementById('btnNext').style.display=n<4?'flex':'none';
  document.getElementById('btnSubmit').style.display=n===4?'flex':'none';
  document.getElementById('stepCounter').textContent='Step '+n+' of 4';
  currentStep=n;
  if(n===4) buildSummary();
  var mb=document.querySelector('#enrollModal .modal-body'); if(mb) mb.scrollTop=0;
  var ov=document.getElementById('enrollModal'); if(ov) ov.scrollTop=0;
  var mdl=document.querySelector('#enrollModal .modal'); if(mdl) mdl.scrollTop=0;
}

/* ── full validation on Next click ── */
function validateStep(n){
  var ok=true;

  function reqField(id){
    var el=document.getElementById(id);
    if(!el) return true;
    if(!el.value.trim()){ setError(el); ok=false; return false; }
    clearError(el); return true;
  }
  function alphaField(id, min, max){
    var el=document.getElementById(id);
    if(!el) return true;
    var v=el.value.trim();
    if(!v){ setError(el); ok=false; return false; }
    if(v.length<(min||2)){ setError(el); ok=false; return false; }
    if(v.length>(max||50)){ setError(el); ok=false; return false; }
    if(!/^[a-zA-Z\u0600-\u06FF\s\-']+$/.test(v)){ setError(el); ok=false; return false; }
    clearError(el); return true;
  }
  function dobField(id){
    var el=document.getElementById(id);
    if(!el) return true;
    // the visible typed field sits inside the .cs-dp wrapper that follows the hidden input
    var wrap=el.nextSibling;
    var typed=(wrap&&wrap.querySelector)?wrap.querySelector('.cs-dp-input'):null;
    var typedTxt=typed?typed.value.trim():'';
    if(el.value.trim()){ clearError(el); return true; }           // valid date committed
    if(typedTxt){ setError(el,'Please enter a valid date (DD/MM/YYYY)'); ok=false; return false; } // typed but invalid
    setError(el,'Please enter date of birth'); ok=false; return false;                              // truly empty
  }
  function phoneField(id){
    var el=document.getElementById(id);
    if(!el) return true;
    var v=el.value.trim();
    if(!v){ setError(el); ok=false; return false; }
    if(v.length<10||v.length>15){ setError(el); ok=false; return false; }
    if(!/^(\+92|0)[0-9\-\s]{9,13}$/.test(v)){ setError(el); ok=false; return false; }
    clearError(el); return true;
  }

  if(n===1){
    alphaField('childFirst',2,30);
    alphaField('childLast',2,30);
    dobField('childDob');
    reqField('childGender');
    reqField('childDiag');
    var checked=document.querySelectorAll('#programsGroup input:checked');
    if(checked.length===0){
      var pErr=document.getElementById('programsErr'); if(pErr) pErr.style.display='block';
      ok=false;
    } else {
      var pErr2=document.getElementById('programsErr'); if(pErr2) pErr2.style.display='none';
    }
  }
  if(n===2){
    alphaField('parentName',3,50);
    reqField('parentRel');
    phoneField('parentPhone');
    reqField('parentArea');
    var emailEl=document.getElementById('parentEmail');
    if(emailEl && emailEl.value.trim().length>0){
      if(!/^[^@\s]+@[^@\s]+\.[^@\s]{2,}$/.test(emailEl.value)){ setError(emailEl); ok=false; }
    }
  }
  if(n===3){
    reqField('prefDate');
    reqField('prefTime');
  }
  return ok;
}

function scrollToFirstError(){
  var active=document.querySelector('#enrollModal .step-panel.active')||document;
  var el=active.querySelector('.error, .cs-dd-btn.error, .cs-dp-input.error');
  if(!el){ var pe=document.getElementById('programsErr'); if(pe && pe.style.display==='block') el=pe; }

function buildSummary(){
  var programs=[].slice.call(document.querySelectorAll('#programsGroup input:checked')).map(function(c){return c.value;}).join(', ')||'';
  var dob=document.getElementById('childDob').value;
  var age='';
  if(dob){ var diff=Date.now()-new Date(dob); age=Math.floor(diff/31557600000)+' years'; }
  var rows=[
    ['Child Name',(document.getElementById('childFirst').value+' '+document.getElementById('childLast').value).trim()],
    ['Age',age],
    ['Gender',document.getElementById('childGender').value],
    ['Condition',document.getElementById('childDiag').value],
    ['Programs',programs],
    ['Parent Name',document.getElementById('parentName').value],
    ['Phone',document.getElementById('parentPhone').value],
    ['City / Area',document.getElementById('parentArea').value],
    ['Date',document.getElementById('prefDate').value||''],
    ['Time',document.getElementById('prefTime').value||'']
  ];
  document.getElementById('summaryBox').innerHTML=
    '<div class="summary-title">&#128203; Your Enrollment Summary</div>'+
    rows.map(function(r){
      return '<div class="summary-row"><span class="summary-label">'+r[0]+'</span><span class="summary-val">'+(r[1]||'')+'</span></div>';
    }).join('');
}

function submitForm(){
  /* prevent double submission (double-tap / slow network / multiple clicks) */
  if(window._nestSubmitting) return;

  if(!document.getElementById('consent').checked){
    var ci=document.getElementById('consentItem');
    ci.style.borderColor='#DC2626';
    if(ci.scrollIntoView) ci.scrollIntoView({behavior:'smooth',block:'center'});
    return;
  }

  /* lock: mark as submitting and disable the button so it can't fire twice */
  window._nestSubmitting=true;
  var _submitBtn=document.getElementById('btnSubmit');
  if(_submitBtn){ _submitBtn.disabled=true; _submitBtn.style.opacity='0.6'; _submitBtn.style.pointerEvents='none'; }

  // Collect all form data
  var programs = [].slice.call(document.querySelectorAll('#programsGroup input:checked'))
                   .map(function(c){ return c.value; }).join(', ');

  var hearAbout = [].slice.call(document.querySelectorAll('#hearGroup .radio-item.selected'))
                    .map(function(c){ return c.dataset.value; }).join(', ');

  var prevTherapy = '';
  var prevChecked = document.querySelector('#prevGroup .radio-item.selected input');
  if(prevChecked) prevTherapy = prevChecked.value;

  var data = {
    childFirst:   document.getElementById('childFirst').value,
    childLast:    document.getElementById('childLast').value,
    dob:          document.getElementById('childDob').value,
    gender:       document.getElementById('childGender').value,
    diagnosis:    document.getElementById('childDiag').value,
    programs:     programs,
    parentName:   document.getElementById('parentName').value,
    relationship: document.getElementById('parentRel').value,
    phone:        document.getElementById('parentPhone').value,
    email:        document.getElementById('parentEmail').value,
    city:         document.getElementById('parentArea').value,
    hearAbout:    hearAbout,
    prefDate:     document.getElementById('prefDate').value,
    prefTime:     document.getElementById('prefTime').value,
    prevTherapy:  prevTherapy,
    notes:        document.getElementById('notes').value
  };

  // Send to Google Sheets
    // Show success screen immediately (independent of network response)
  function _nestShowSuccess(){
    document.querySelectorAll('#enrollModal .step-panel').forEach(function(p){ p.classList.remove('active'); });
    document.getElementById('successScreen').classList.add('show');
    if(typeof gtag==='function'){ gtag('event','generate_lead',{event_category:'enrollment',event_label:'Enrollment Form Submitted'}); }
    document.getElementById('modalFooter').style.display='none';
    document.getElementById('stepDots').style.display='none';
    if(window._nestResetTimer) clearTimeout(window._nestResetTimer);
    window._nestResetTimer = setTimeout(function(){ closeEnrollModal(); }, 4000);
  }
  fetch('https://script.google.com/macros/s/AKfycbz_njfYUo78Eo2q-TKhFVA8EKZrPItRH1uEZGYzr9y5o0xB2bgS7NFHXOQDPDrUPc7ltA/exec', {
    method: 'POST',
    body: JSON.stringify(data)
  }).catch(function(err){ console.error(err); });
  _nestShowSuccess();
}
